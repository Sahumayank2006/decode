133: function safeNumber(value: unknown, fallback = 0): number {
134:   if (typeof value === "number") {
135:     return Number.isFinite(value) ? value : fallback;
136:   }
137: 
138:   if (typeof value === "string") {
139:     const cleaned = value
140:       .replace(/,/g, "")
141:       .replace(/%/g, "")
142:       .trim();
143: 
144:     const parsed = Number(cleaned);
145: 
146:     return Number.isFinite(parsed)
147:       ? parsed
148:       : fallback;
149:   }
150: 
151:   if (
152:     value &&
153:     typeof value === "object"
154:   ) {
155:     const objectValue = value as Record<string, unknown>;
156: 
157:     return safeNumber(
158:       objectValue.value ??
159:         objectValue.y ??
160:         objectValue.numeric_value ??
161:         objectValue.number,
162:       fallback
163:     );
164:   }
165: 
166:   return fallback;
167: }
168: 
169: function cleanLabel(value: unknown, fallback: string): string {
170:   if (
171:     typeof value === "string" &&
172:     value.trim()
173:   ) {
174:     return value.trim();
175:   }
176: 
177:   if (
178:     typeof value === "number" &&
179:     Number.isFinite(value)
180:   ) {
181:     return String(value);
182:   }
183: 
184:   return fallback;
185: }

207:   if (
208:     normalized === "line" ||
209:     normalized.includes("line")

216:     normalized.includes("area")

351: function normalizeFromDataRows(
352:   chart: any
353: ): {
354:   categories: string[];
355:   series: ExtractedSeries[];
356: } | null {
357:   const rows =
358:     Array.isArray(chart?.data)
359:       ? chart.data
360:       : Array.isArray(chart?.rows)
361:         ? chart.rows
362:         : null;
363: 
364:   if (!rows || rows.length === 0) {
365:     return null;
366:   }
367: 
368:   if (
369:     typeof rows[0] !== "object" ||
370:     rows[0] === null
371:   ) {
372:     return null;
373:   }
374: 
375:   const firstRow =
376:     rows[0] as Record<string, unknown>;
377: 
378:   const keys = Object.keys(firstRow);
379: 
380:   const categoryKey =
381:     [
382:       "category",
383:       "label",
384:       "name",
385:       "x",
386:       "group",
387:     ].find((key) =>
388:       keys.includes(key)
389:     ) ?? keys[0];
390: 
391:   const numericKeys = keys.filter(
392:     (key) => {
393:       if (key === categoryKey) {
394:         return false;
395:       }
396: 
397:       return rows.some((row: any) => {
398:         const value = row?.[key];
399: 
400:         if (
401:           typeof value === "number" &&
402:           Number.isFinite(value)
403:         ) {
404:           return true;
405:         }
406: 
407:         if (
408:           typeof value === "string" &&
409:           value.trim() !== ""
410:         ) {
411:           return Number.isFinite(
412:             Number(
413:               value.replace(/,/g, "")
414:             )
415:           );
416:         }
417: 
418:         return false;
419:       });
420:     }
421:   );
422: 
423:   if (numericKeys.length === 0) {
424:     return null;
425:   }
426: 
427:   const categories = rows.map(
428:     (row: any, index: number) =>
429:       cleanLabel(
430:         row?.[categoryKey],
431:         `Category ${index + 1}`
432:       )
433:   );
434: 
435:   const series = numericKeys.map(
436:     (key, seriesIndex) => ({
437:       id: `series-${seriesIndex}-${key}`,
438:       name: key,
439:       values: rows.map(
440:         (row: any) =>
441:           safeNumber(row?.[key])
442:       ),
443:     })
444:   );
445: 
446:   return {
447:     categories,
448:     series,
449:   };
450: }

456: function normalizeExtractedChart(
457:   payload: any,
458:   fallbackIndex = 0
459: ): ExtractedChart | null {
460:   const chart = unwrapChart(payload);
461: 
462:   if (!chart) {
463:     return null;
464:   }
465: 
466:   const id = String(
467:     chart.id ??
468:       chart.chart_id ??
469:       chart.chartId ??
470:       `extracted-chart-${fallbackIndex + 1}`
471:   );
472: 
473:   const title =
474:     cleanLabel(
475:       chart.title ??
476:         chart.name ??
477:         chart.chart_title ??
478:         chart.figure_title,
479:       `Extracted Chart ${fallbackIndex + 1}`
480:     );
481: 
482:   const chartType = normalizeChartType(
483:     chart.chart_type ??
484:       chart.chartType ??
485:       chart.type ??
486:       chart.render_type ??
487:       chart.renderType
488:   );
489: 
490:   const confidenceRaw =
491:     chart.confidence ??
492:     chart.source_confidence ??
493:     chart.extraction_confidence ??
494:     chart.score ??
495:     null;
496: 
497:   let confidence: number | null = null;
498: 
499:   if (confidenceRaw !== null) {
500:     const number = safeNumber(
501:       confidenceRaw,
502:       NaN
503:     );
504: 
505:     if (Number.isFinite(number)) {
506:       confidence =
507:         number <= 1
508:           ? number * 100
509:           : number;
510:     }
511:   }
512: 
513:   let categories: string[] =
514:     Array.isArray(chart.categories)
515:       ? chart.categories.map(
516:           (item: unknown, index: number) =>
517:             cleanLabel(
518:               item,
519:               `Category ${index + 1}`
520:             )
521:         )
522:       : Array.isArray(chart.labels)
523:         ? chart.labels.map(
524:             (item: unknown, index: number) =>
525:               cleanLabel(
526:                 item,
527:                 `Category ${index + 1}`
528:               )
529:           )
530:         : Array.isArray(chart.x_values)
531:           ? chart.x_values.map(
532:               (item: unknown, index: number) =>
533:                 cleanLabel(
534:                   item,
535:                   `Category ${index + 1}`
536:                 )
537:             )
538:           : [];
539: 
540:   let series: ExtractedSeries[] = [];
541: 
542:   if (
543:     Array.isArray(chart.series) &&
544:     chart.series.length > 0
545:   ) {
546:     series = chart.series.map(
547:       (item: any, index: number) => ({
548:         id: String(
549:           item?.id ??
550:             item?.series_id ??
551:             `series-${index}`
552:         ),
553:         name: cleanLabel(
554:           item?.name ??
555:             item?.label ??
556:             item?.title ??
557:             item?.key,
558:           `Series ${index + 1}`
559:         ),
560:         values: getSeriesValues(item),
561:       })
562:     );
563: 
564:     if (
565:       categories.length === 0
566:     ) {
567:       categories =
568:         inferCategoriesFromSeries(
569:           chart.series
570:         );
571:     }
572:   }
573: 
574:   if (
575:     categories.length === 0 ||
576:     series.length === 0
577:   ) {
578:     const fromRows =
579:       normalizeFromDataRows(chart);
580: 
581:     if (fromRows) {
582:       categories =
583:         fromRows.categories;
584: 
585:       series =
586:         fromRows.series;
587:     }
588:   }
589: 
590:   /*
591:    * Some backend formats store:
592:    *
593:    * data: {
594:    *   categories: [],
595:    *   series: []
596:    * }
597:    */
598:   if (
599:     categories.length === 0 &&
600:     chart.data &&
601:     typeof chart.data === "object" &&
602:     !Array.isArray(chart.data)
603:   ) {
604:     const nested =
605:       chart.data;
606: 
607:     if (
608:       Array.isArray(
609:         nested.categories
610:       )
611:     ) {
612:       categories =
613:         nested.categories.map(
614:           (
615:             item: unknown,
616:             index: number
617:           ) =>
618:             cleanLabel(
619:               item,
620:               `Category ${index + 1}`
621:             )
622:         );
623:     }
624: 
625:     if (
626:       Array.isArray(nested.series)
627:     ) {
628:       series =
629:         nested.series.map(
630:           (
631:             item: any,
632:             index: number
633:           ) => ({
634:             id: String(
635:               item?.id ??
636:                 `series-${index}`
637:             ),
638:             name: cleanLabel(
639:               item?.name ??
640:                 item?.label,
641:               `Series ${index + 1}`
642:             ),
643:             values:
644:               getSeriesValues(item),
645:           })
646:         );
647:     }
648:   }
649: 
650:   /*
651:    * Never pretend an empty extraction
652:    * is valid chart data.
653:    */
654:   if (
655:     categories.length === 0 ||
656:     series.length === 0
657:   ) {
658:     return {
659:       id,
660:       title,
661:       chartType,
662:       confidence,
663:       categories: [],
664:       series: [],
665:       page:
666:         chart.page ??
667:         chart.page_number ??
668:         chart.pageNumber ??
669:         null,
670:       raw: payload,
671:     };
672:   }
673: 
674:   const maxLength =
675:     Math.max(
676:       categories.length,
677:       ...series.map(
678:         (item) =>
679:           item.values.length
680:       )
681:     );
682: 
683:   while (
684:     categories.length <
685:     maxLength
686:   ) {
687:     categories.push(
688:       `Category ${categories.length + 1}`
689:     );
690:   }
691: 
692:   series = series.map(
693:     (item) => ({
694:       ...item,
695:       values: [
696:         ...item.values,
697:         ...Array(
698:           Math.max(
699:             0,
700:             categories.length -
701:               item.values.length
702:           )
703:         ).fill(0),
704:       ],
705:     })
706:   );
707: 
708:   return {
709:     id,
710:     title,
711:     chartType,
712:     confidence,
713:     categories,
714:     series,
715:     page:
716:       chart.page ??
717:       chart.page_number ??
718:       chart.pageNumber ??
719:       null,
720:     raw: payload,
721:   };
722: }
